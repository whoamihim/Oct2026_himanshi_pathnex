DAY 02 - Services & Tags

Ansible Task - Install and Enable Nginx

---
- name: Setup Nginx on my server
  hosts: all
  become: yes

  tasks:
    - name: Nginx package install
      yum:
        name: nginx
        state: present

    - name: Nginx service start and enable on boot
      service:
        name: nginx
        state: started
        enabled: yes


Terraform Task - EC2 with Tags (t3.micro)

provider "aws" {
  region = "ap-south-1"
}

resource "aws_instance" "himanshi_web" {
  ami           = "ami-0abcd1234abcd1234"
  instance_type = "t3.micro"

  tags = {
    Name        = "himanshi-web-server"
    Environment = "Practice"
    Owner       = "Himanshi"
  }
}


Kubernetes Task - Deployment with 2 Replicas

apiVersion: apps/v1
kind: Deployment
metadata:
  name: himanshi-nginx-deploy
spec:
  replicas: 2
  selector:
    matchLabels:
      app: himanshi-web
  template:
    metadata:
      labels:
        app: himanshi-web
    spec:
      containers:
        - name: web
          image: nginx
          ports:
            - containerPort: 80

Jenkins Pipeline - Maven Build

pipeline {
    agent any
    tools {
        maven 'Maven-3.8.1'
        jdk 'JDK-17'
    }
    stages {
        stage('Get Code') {
            steps {
                git url: 'https://github.com/Pathnex/sample-java-app.git'
            }
        }
        stage('Build') {
            steps {
                sh 'mvn clean compile'
            }
        }
        stage('Unit Test') {
            steps {
                sh 'mvn test'
            }
        }
        stage('Create JAR') {
            steps {
                sh 'mvn package'
            }
        }
    }
}


GitLab CI - Maven Build
