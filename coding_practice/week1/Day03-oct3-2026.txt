DAY 03 - Users, Security Groups, Env Vars

Ansible Task - Create User

---
- name: Create my admin user
  hosts: all
  become: yes

  tasks:
    - name: Create user himanshi
      user:
        name: himanshi
        shell: /bin/bash


Terraform Task - Security Group with EC2 (t3.micro)

provider "aws" {
  region = "ap-south-1"
}

resource "aws_security_group" "himanshi_sg" {
  name        = "himanshi-ssh-sg"
  description = "Allow SSH to my server"

  ingress {
    from_port   = 22
    to_port     = 22
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }
}

resource "aws_instance" "himanshi_web" {
  ami                    = "ami-0abcd1234abcd1234"
  instance_type          = "t3.micro"
  vpc_security_group_ids = [aws_security_group.himanshi_sg.id]

  tags = {
    Name = "himanshi-ec2"
  }
}


Kubernetes Task - Pod with Environment Variables

apiVersion: v1
kind: Pod
metadata:
  name: himanshi-env-pod
spec:
  containers:
    - name: web
      image: nginx
      env:
        - name: APP_ENV
          value: "practice"
        - name: OWNER
          value: "himanshi"


Shell Script - Check File Exists

#!/bin/bash
FILE="/tmp/himanshi.txt"

if [ -f "$FILE" ]; then
  echo "File exists"
else
  echo "File does not exist"
fi


Jenkins Pipeline - NodeJS Build

pipeline {
    agent any
    tools {
        nodejs 'NodeJS-16'
    }
    stages {
        stage('Get Code') {
            steps {
                git url: 'https://github.com/Pathnex/sample-node-app.git'
            }
        }
        stage('Install Packages') {
            steps {
                sh 'npm install'
            }
        }
        stage('Test') {
            steps {
                sh 'npm test'
            }
        }
        stage('Build') {
            steps {
                sh 'npm run build'
            }
        }
    }
}
